For `getById` and `getWithOrgLink`, the authorization check must verify a partnership exists:
```typescript
export const getById = query({
  args: {
    haulerId: v.id("haulers"),
    treaterId: v.id("treaters"),  // NEW: required for auth
  },
  handler: async (ctx, args) => {
    await requireAuth(ctx);  // NEW: require authentication

    const hauler = await ctx.db.get(args.haulerId);
    if (!hauler) return null;

    // NEW: Verify partnership exists between hauler and treater
    const partnership = await ctx.db
      .query("treaterHaulerPartners")
      .withIndex("by_treater", (q) => q.eq("treaterId", args.treaterId))
      .collect()
      .then(partnerships => partnerships.find(p => p.haulerId === args.haulerId && p.isActive));

    if (!partnership) {
      throw new ConvexError("Access denied: No active partnership with this hauler");
    }

    return hauler;
  },
});
```

Note: Haulers don't have direct treaterId - access is through treaterHaulerPartners table.
