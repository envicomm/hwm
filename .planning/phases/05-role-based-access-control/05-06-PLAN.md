1. Create usePermissions hook (apps/treater/src/hooks/usePermissions.ts):

```typescript
import { useQuery } from "@tanstack/react-query";
import { authClient } from "@/lib/auth";

export type OrgRole = "owner" | "admin" | "member";

// Permission resource-action matrix (mirrors server-side PERMISSIONS)
const PERMISSIONS = {
  organization: {
    update: ["owner", "admin"] as OrgRole[],
    delete: ["owner"] as OrgRole[],
    manageBilling: ["owner"] as OrgRole[],
  },
  team: {
    invite: ["owner", "admin"] as OrgRole[],
    remove: ["owner", "admin"] as OrgRole[],
    updateRole: ["owner", "admin"] as OrgRole[],
    listMembers: ["owner", "admin", "member"] as OrgRole[],
  },
  generator: {
    create: ["owner", "admin"] as OrgRole[],
    update: ["owner", "admin"] as OrgRole[],
    delete: ["owner"] as OrgRole[],
    view: ["owner", "admin", "member"] as OrgRole[],
  },
  hauler: {
    create: ["owner", "admin"] as OrgRole[],
    update: ["owner", "admin"] as OrgRole[],
    delete: ["owner"] as OrgRole[],
    view: ["owner", "admin", "member"] as OrgRole[],
  },
  wasteBag: {
    create: ["owner", "admin", "member"] as OrgRole[],
    update: ["owner", "admin"] as OrgRole[],
    delete: ["owner"] as OrgRole[],
    view: ["owner", "admin", "member"] as OrgRole[],
  },
} as const;

type PermissionResource = keyof typeof PERMISSIONS;
type PermissionAction<R extends PermissionResource> = keyof (typeof PERMISSIONS)[R];

function hasPermission<R extends PermissionResource>(
  role: OrgRole | undefined,
  resource: R,
  action: PermissionAction<R>
): boolean {
  if (!role) return false;
  const resourcePerms = PERMISSIONS[resource];
  const allowedRoles = resourcePerms[action as keyof typeof resourcePerms] as readonly OrgRole[];
  return allowedRoles?.includes(role) ?? false;
}

/**
 * Hook to check user's permissions in the active organization
 * Returns current role and permission check functions
 */
export function usePermissions() {
  const { data: activeMember, isPending, error } = useQuery({
    queryKey: ["active-member"],
    queryFn: async () => {
      const result = await authClient.organization.getActiveMember({});
      return result.data;
    },
    staleTime: 30000, // Cache for 30 seconds
  });

  const role = activeMember?.role as OrgRole | undefined;

  return {
    role,
    isPending,
    error,
    isOwner: role === "owner",
    isAdmin: role === "admin" || role === "owner",
    isMember: role === "member",

    /**
     * Check if user can perform action on resource
     */
    can: <R extends PermissionResource>(resource: R, action: PermissionAction<R>): boolean => {
      return hasPermission(role, resource, action);
    },

    /**
     * Check if user has one of the required roles
     */
    hasRole: (...roles: OrgRole[]): boolean => {
      return role ? roles.includes(role) : false;
    },
  };
}
```

2. Create PermissionGate component (apps/treater/src/components/ui/permission-gate.tsx):

```typescript
import type { ReactNode } from "react";
import { usePermissions, type OrgRole } from "@/hooks/usePermissions";

type PermissionResource = "organization" | "team" | "generator" | "hauler" | "wasteBag";

interface PermissionGateProps {
  children: ReactNode;
  /** Fallback to render when permission denied (default: null) */
  fallback?: ReactNode;
  /** Required roles - user must have one of these */
  requiredRoles?: OrgRole[];
  /** Permission check - resource and action */
  permission?: {
    resource: PermissionResource;
    action: string;
  };
}

/**
 * Conditionally renders children based on user's role/permissions
 *
 * Usage:
 * <PermissionGate requiredRoles={["owner", "admin"]}>
 *   <Button>Admin Action</Button>
 * </PermissionGate>
 *
 * <PermissionGate permission={{ resource: "generator", action: "delete" }}>
 *   <Button variant="destructive">Delete</Button>
 * </PermissionGate>
 */
export function PermissionGate({
  children,
  fallback = null,
  requiredRoles,
  permission,
}: PermissionGateProps) {
  const { role, can, isPending } = usePermissions();

  // While loading, don't render anything to prevent flash
  if (isPending) {
    return null;
  }

  // Check role requirement
  if (requiredRoles && role) {
    if (!requiredRoles.includes(role)) {
      return <>{fallback}</>;
    }
  }

  // Check permission requirement
  if (permission) {
    // Type assertion needed since action is string
    if (!can(permission.resource, permission.action as never)) {
      return <>{fallback}</>;
    }
  }

  return <>{children}</>;
}
```
