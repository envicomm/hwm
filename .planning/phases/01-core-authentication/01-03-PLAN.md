  <action>
    Update the auth page to:
    1. Redirect authenticated users away from auth pages to /dashboard
    2. Handle post-authentication redirect to /dashboard
    3. Configure AuthView with proper callbacks

    Add beforeLoad to check auth and redirect:
    ```typescript
    import { createFileRoute, redirect, useNavigate } from "@tanstack/react-router"
    import { AuthView } from "@daveyplate/better-auth-ui"
    // ... existing imports

    export const Route = createFileRoute("/auth/$authView")({
      beforeLoad: async ({ context }) => {
        // If already authenticated, redirect to dashboard
        if (context.isAuthenticated) {
          throw redirect({ to: "/dashboard" })
        }
      },
      component: AuthPage,
    })

    function AuthPage() {
      const { authView } = Route.useParams()
      const navigate = useNavigate()

      // Handle successful auth - redirect to dashboard
      const handleAuthSuccess = () => {
        navigate({ to: "/dashboard" })
      }

      return (
        // ... existing layout JSX
        <AuthView
          pathname={authView}
          onAuthSuccess={handleAuthSuccess}
          // AuthView handles sign-in, sign-up, forgot-password, reset-password views
        />
        // ... rest of layout
      )
    }
    ```

    Note: The exact callback prop may be `onAuthSuccess`, `onSuccess`, or similar - check @daveyplate/better-auth-ui docs. If no callback exists, use useEffect to watch auth state and redirect.

    Alternative pattern if no callback:
    ```typescript
    import { useConvexAuth } from "convex/react"

    function AuthPage() {
      const { isAuthenticated } = useConvexAuth()
      const navigate = useNavigate()

      useEffect(() => {
        if (isAuthenticated) {
          navigate({ to: "/dashboard" })
        }
      }, [isAuthenticated, navigate])

      // ... rest of component
    }
    ```
