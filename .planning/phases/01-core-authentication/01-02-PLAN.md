    import { createRootRouteWithContext, Outlet, HeadContent, Scripts } from "@tanstack/react-router"
    import { ConvexBetterAuthProvider } from "@convex-dev/better-auth/react"
    import { getToken } from "~/lib/auth-server"

    // Server function to get auth token
    const getAuth = createServerFn({ method: 'GET' }).handler(
      async () => await getToken()
    )

    export const Route = createRootRouteWithContext<{
      queryClient: QueryClient
    }>()({
      beforeLoad: async ({ context }) => {
        const token = await getAuth()
        // If using ConvexQueryClient, set server auth
        if (token && context.convexQueryClient?.serverHttpClient) {
          context.convexQueryClient.serverHttpClient.setAuth(token)
        }
        return { isAuthenticated: !!token, token }
      },
      component: RootComponent,
      head: () => ({
        meta: [
          { charSet: "utf-8" },
          { name: "viewport", content: "width=device-width, initial-scale=1" },
          { title: "HWM Treater" },

<task type="auto">
  <name>Task 3: Replace mock auth context with Better Auth integration</name>
  <files>apps/treater/src/contexts/auth-context.tsx</files>
  <action>
    Replace the mock localStorage-based auth with real Better Auth session integration.

    Key patterns:
    1. Use `useConvexAuth()` from convex/react for auth state (NOT Better Auth's useSession)
    2. Use authClient for sign-in/sign-out actions
    3. Use Convex query to get user details

    Implementation:
    ```typescript
    import { createContext, useContext, type ReactNode } from "react"
    import { useConvexAuth } from "convex/react"
    import { useQuery } from "convex/react"
    import { api } from "@hwm/convex"
    import { authClient } from "~/lib/auth"

    interface AuthUser {
      id: string
      email: string
      name: string | null
      emailVerified: boolean
      image?: string | null
    }

    interface AuthContextType {
      user: AuthUser | null
      isAuthenticated: boolean
      isLoading: boolean
      signIn: typeof authClient.signIn
      signUp: typeof authClient.signUp
      signOut: typeof authClient.signOut
    }

    const AuthContext = createContext<AuthContextType | null>(null)

    export function AuthProvider({ children }: { children: ReactNode }) {
      // Use Convex's auth state - this is the source of truth
      const { isAuthenticated, isLoading: authLoading } = useConvexAuth()

      // Get user from Convex (only when authenticated)
      const authUser = useQuery(
        api.auth.getCurrentUser,
        isAuthenticated ? {} : "skip"
      )

      const isLoading = authLoading || (isAuthenticated && authUser === undefined)

      const user: AuthUser | null = authUser ? {
        id: authUser.id,
        email: authUser.email,
        name: authUser.name,
        emailVerified: authUser.emailVerified,
        image: authUser.image,
      } : null

      return (
        <AuthContext.Provider
          value={{
            user,
            isAuthenticated,
            isLoading,
            signIn: authClient.signIn,
            signUp: authClient.signUp,
            signOut: authClient.signOut,
          }}
        >
          {children}
        </AuthContext.Provider>
      )
    }

    export function useAuth() {
      const context = useContext(AuthContext)
      if (!context) {
        throw new Error("useAuth must be used within an AuthProvider")
      }
      return context
    }
    ```

    IMPORTANT: Use `useConvexAuth()` NOT `authClient.useSession()` - Better Auth reflects authenticated state before Convex validates the token, causing race conditions.
